"""
LNMO (LiNi0.5Mn1.5O4) 3-phase Powder XRD Simulator — v2 (Benchtop Cu Ka1/Ka2, Measured Conditions).

v2 calibration against A-Lab Precursor Genome benchtop Cu Ka scans:
  - 2-min routine & 10-min precision scan counting statistics
  - Flat Mn fluorescence background scaled to spinel (400) height (`bg_over_400`)
  - Sample-height displacement error (`displacement_mm`, R = 150 mm, delta2theta = -2*s*cos(theta)/R)
  - Benchtop Mn-spinel instrumental broadening (u_inst=0.009, w_inst=0.0085)
  - Preserves exact v1 crystallography, structure factors, area-normalized pseudo-Voigt, Ka1/Ka2 doublet
"""

from __future__ import annotations

import math
from dataclasses import dataclass
from typing import Dict, List, Tuple

import numpy as np

LAMBDA_KA1 = 1.54056
LAMBDA_KA2 = 1.54439
KA2_RATIO = 0.5

CROMER_MANN = {
    "Li+": (0.6968, 4.6237, 0.7888, 1.9557, 0.3414, 0.6316, 0.1563, 10.0953, 0.0167),
    "Ni2+": (11.4166, 3.6766, 7.4005, 0.2449, 5.3442, 8.8730, 0.9773, 22.1626, 0.8614),
    "Mn4+": (10.8061, 4.1329, 6.2323, 0.2584, 3.0958, 10.5060, 0.2202, 27.6001, 0.6461),
    "O2-": (3.0485, 13.2771, 2.2868, 5.7011, 1.5463, 0.3239, 0.8670, 32.9089, 0.2508),
}

ANOMALOUS_CU = {
    "Li+": (0.000, 0.000),
    "Ni2+": (-3.003, 0.509),
    "Mn4+": (-0.530, 2.805),
    "O2-": (0.049, 0.032),
}

ATOMIC_MASS = {
    "Li+": 6.941,
    "Ni2+": 58.6934,
    "Mn4+": 54.9380,
    "O2-": 15.9994,
}


def disp_deg(pos: np.ndarray, s_mm: float, r_mm: float = 150.0) -> np.ndarray:
    """Bragg-Brentano sample-height displacement shift in 2theta (degrees)."""
    if abs(s_mm) < 1e-12:
        return np.zeros_like(pos)
    return -np.degrees(2.0 * s_mm * np.cos(np.radians(pos / 2.0)) / r_mm)


def atomic_form_factor(ion: str, s: np.ndarray) -> np.ndarray:
    a1, b1, a2, b2, a3, b3, a4, b4, c = CROMER_MANN[ion]
    s2 = s * s
    f0 = (
        a1 * np.exp(-b1 * s2)
        + a2 * np.exp(-b2 * s2)
        + a3 * np.exp(-b3 * s2)
        + a4 * np.exp(-b4 * s2)
        + c
    )
    fp, fpp = ANOMALOUS_CU[ion]
    return (f0 + fp) + 1j * fpp


def _unique_frac_coords(coords: List[Tuple[float, float, float]], tol: float = 1e-4) -> np.ndarray:
    unique: List[np.ndarray] = []
    for c in coords:
        v = np.mod(np.array(c, dtype=float), 1.0)
        v[np.isclose(v, 1.0, atol=tol)] = 0.0
        is_dup = False
        for u in unique:
            diff = np.abs(v - u)
            diff = np.minimum(diff, 1.0 - diff)
            if np.all(diff < tol):
                is_dup = True
                break
        if not is_dup:
            unique.append(v)
    return np.array(unique, dtype=float)


def _apply_fcc_translations(coords: List[Tuple[float, float, float]]) -> List[Tuple[float, float, float]]:
    shifts = [(0.0, 0.0, 0.0), (0.0, 0.5, 0.5), (0.5, 0.0, 0.5), (0.5, 0.5, 0.0)]
    out = []
    for x, y, z in coords:
        for sx, sy, sz in shifts:
            out.append((x + sx, y + sy, z + sz))
    return out


def _generate_spacegroup_ops(generators: List[Tuple[np.ndarray, np.ndarray]]) -> List[Tuple[np.ndarray, np.ndarray]]:
    ops: List[Tuple[np.ndarray, np.ndarray]] = [(np.eye(3, dtype=float), np.zeros(3, dtype=float))]

    def _same(op1, op2):
        if not np.allclose(op1[0], op2[0], atol=1e-6):
            return False
        dt = np.mod(op1[1] - op2[1], 1.0)
        dt = np.minimum(dt, 1.0 - dt)
        return np.all(dt < 1e-5)

    queue = list(ops)
    while queue:
        R1, t1 = queue.pop(0)
        for R2, t2 in generators:
            R_new = R2 @ R1
            t_new = np.mod(R2 @ t1 + t2, 1.0)
            cand = (R_new, t_new)
            if not any(_same(cand, existing) for existing in ops):
                ops.append(cand)
                queue.append(cand)
    return ops


_P4332_GENERATORS = [
    (np.array([[0, 0, 1], [1, 0, 0], [0, 1, 0]], dtype=float), np.array([0.0, 0.0, 0.0])),
    (np.array([[-1, 0, 0], [0, -1, 0], [0, 0, 1]], dtype=float), np.array([0.5, 0.0, 0.5])),
    (np.array([[0, -1, 0], [-1, 0, 0], [0, 0, -1]], dtype=float), np.array([0.25, 0.25, 0.25])),
]
_P4332_OPS = _generate_spacegroup_ops(_P4332_GENERATORS)
assert len(_P4332_OPS) == 24


def _apply_p4332(xyz: Tuple[float, float, float]) -> np.ndarray:
    v = np.array(xyz, dtype=float)
    pts = [tuple(R @ v + t) for R, t in _P4332_OPS]
    return _unique_frac_coords(pts)


def build_fd3m_sites(
    u_o: float = 0.263,
    tm_8a_occ: float = 0.0,
    li_ratio: float = 1.0,
    ni_ratio: float = 0.5,
    mn_ratio: float = 1.5,
) -> List[Tuple[str, float, float, np.ndarray]]:
    li_coords = _unique_frac_coords(_apply_fcc_translations([(1 / 8, 1 / 8, 1 / 8), (7 / 8, 7 / 8, 7 / 8)]))
    tm_coords = _unique_frac_coords(
        _apply_fcc_translations([(0.5, 0.5, 0.5), (0.5, 0.25, 0.25), (0.25, 0.5, 0.25), (0.25, 0.25, 0.5)])
    )
    u = u_o
    o_coords = _unique_frac_coords(
        _apply_fcc_translations(
            [
                (u, u, u),
                (u, 0.25 - u, 0.25 - u),
                (0.25 - u, u, 0.25 - u),
                (0.25 - u, 0.25 - u, u),
                (-u, -u, -u),
                (-u, u - 0.25, u - 0.25),
                (u - 0.25, -u, u - 0.25),
                (u - 0.25, u - 0.25, -u),
            ]
        )
    )

    tm_8a = float(np.clip(tm_8a_occ, 0.0, 0.08))
    li_excess = max(0.0, li_ratio - 1.0)
    li_8a = float(np.clip(min(li_ratio, 1.0) - tm_8a, 0.85, 1.0))
    tm_sum = max(ni_ratio + mn_ratio, 1.0)
    ni_frac_tm = ni_ratio / tm_sum
    mn_frac_tm = mn_ratio / tm_sum

    li_16d = float(np.clip(0.5 * (li_excess + tm_8a), 0.0, 0.08))
    rem_16d = 1.0 - li_16d
    ni_16d = float(np.clip(rem_16d * ni_frac_tm, 0.15, 0.35))
    mn_16d = float(np.clip(rem_16d * mn_frac_tm, 0.60, 0.85))

    sites = [
        ("Li+", li_8a, 1.2, li_coords),
        ("Ni2+", ni_16d, 0.65, tm_coords),
        ("Mn4+", mn_16d, 0.65, tm_coords),
        ("O2-", 1.0, 0.85, o_coords),
    ]
    if tm_8a > 1e-6:
        sites.append(("Ni2+", tm_8a, 0.8, li_coords))
    if li_16d > 1e-6:
        sites.append(("Li+", li_16d, 1.2, tm_coords))
    return sites


def build_p4332_sites(
    u_shift: float = 0.0,
    order_degree: float = 1.0,
    li_ratio: float = 1.0,
    ni_ratio: float = 0.5,
    mn_ratio: float = 1.5,
) -> List[Tuple[str, float, float, np.ndarray]]:
    li_coords = _apply_p4332((0.0003, 0.0003, 0.0003))
    ni_coords = _apply_p4332((0.625, 0.625, 0.625))
    y_mn = 0.3728 + 0.0012 * (1.0 - order_degree)
    mn_coords = _apply_p4332((0.125, y_mn, 0.25 - y_mn))
    o1_coords = _apply_p4332((0.3872 + u_shift, 0.3872 + u_shift, 0.3872 + u_shift))
    o2_coords = _apply_p4332((0.3892 + u_shift, 0.1112 - u_shift, 0.1130))

    od = float(np.clip(order_degree, 0.3, 1.0))
    tm_sum = max(ni_ratio + mn_ratio, 1.0)
    ni_total_16d = 2.0 * (ni_ratio / tm_sum)
    ni_on_4b = float(
        np.clip((ni_total_16d / 2.0) + (1.0 - ni_total_16d / 2.0) * od * min(1.0, ni_total_16d / 0.5), 0.25, 1.0)
    )
    mn_on_4b = max(0.0, 1.0 - ni_on_4b)
    ni_on_12d = float(np.clip((ni_total_16d - 0.5 * ni_on_4b) / 1.5, 0.0, 0.5))
    mn_on_12d = max(0.0, 1.0 - ni_on_12d)
    li_8c = float(np.clip(min(li_ratio, 1.0), 0.88, 1.0))

    sites = [
        ("Li+", li_8c, 1.1, li_coords),
        ("Ni2+", ni_on_4b, 0.5, ni_coords),
        ("Mn4+", mn_on_12d, 0.5, mn_coords),
        ("O2-", 1.0, 0.75, o1_coords),
        ("O2-", 1.0, 0.75, o2_coords),
    ]
    if mn_on_4b > 1e-6:
        sites.append(("Mn4+", mn_on_4b, 0.5, ni_coords))
    if ni_on_12d > 1e-6:
        sites.append(("Ni2+", ni_on_12d, 0.5, mn_coords))
    return sites


def build_rocksalt_sites(li_fraction: float = 0.22) -> List[Tuple[str, float, float, np.ndarray]]:
    cation_coords = _unique_frac_coords(_apply_fcc_translations([(0.0, 0.0, 0.0)]))
    anion_coords = _unique_frac_coords(_apply_fcc_translations([(0.5, 0.5, 0.5)]))
    x = float(np.clip(li_fraction, 0.05, 0.40))
    return [
        ("Li+", x, 1.0, cation_coords),
        ("Ni2+", 1.0 - x, 0.7, cation_coords),
        ("O2-", 1.0, 0.9, anion_coords),
    ]


def _cubic_reflections(a: float, max_2theta: float, wavelength: float) -> Tuple[np.ndarray, np.ndarray, np.ndarray]:
    sin_max = math.sin(math.radians(max_2theta / 2.0))
    d_min = wavelength / (2.0 * sin_max)
    max_h = int(math.floor(a / d_min)) + 1

    grid = np.arange(-max_h, max_h + 1)
    hh, kk, ll = np.meshgrid(grid, grid, grid, indexing="ij")
    hkl = np.stack([hh.ravel(), kk.ravel(), ll.ravel()], axis=1)
    h2k2l2 = np.sum(hkl * hkl, axis=1)
    mask = (h2k2l2 > 0) & (h2k2l2 <= (a / d_min) ** 2 + 1e-9)
    hkl = hkl[mask]
    h2k2l2 = h2k2l2[mask]
    d = a / np.sqrt(h2k2l2)
    return hkl, h2k2l2, d


def _march_dollase_factor(hkl: np.ndarray, r: float, axis: Tuple[float, float, float] = (1.0, 1.0, 1.0)) -> np.ndarray:
    if abs(r - 1.0) < 1e-5:
        return np.ones(len(hkl), dtype=float)
    ax = np.array(axis, dtype=float)
    ax = ax / np.linalg.norm(ax)
    norms = np.linalg.norm(hkl, axis=1)
    cos_a = np.abs((hkl @ ax) / norms)
    sin_a_sq = np.maximum(1.0 - cos_a * cos_a, 0.0)
    return (r * r * cos_a * cos_a + sin_a_sq / r) ** (-1.5)


def compute_phase_peaks(
    sites: List[Tuple[str, float, float, np.ndarray]],
    a: float,
    min_2theta: float = 10.0,
    max_2theta: float = 90.0,
    wavelength: float = LAMBDA_KA1,
    pref_orient_r: float = 1.0,
) -> Tuple[np.ndarray, np.ndarray, float]:
    hkl, h2k2l2, d = _cubic_reflections(a, max_2theta + 2.0, wavelength)
    sin_theta = wavelength / (2.0 * d)
    valid = (sin_theta > 0) & (sin_theta < 1.0)
    hkl, h2k2l2, d, sin_theta = hkl[valid], h2k2l2[valid], d[valid], sin_theta[valid]
    two_theta = np.degrees(2.0 * np.arcsin(sin_theta))

    in_range = (two_theta >= min_2theta - 2.0) & (two_theta <= max_2theta + 2.0)
    hkl, h2k2l2, d, sin_theta = hkl[in_range], h2k2l2[in_range], d[in_range], sin_theta[in_range]

    s = sin_theta / wavelength
    s2 = s * s

    F = np.zeros(len(hkl), dtype=np.complex128)
    cell_mass_amu = 0.0
    for ion, occ, b_iso, coords in sites:
        cell_mass_amu += occ * len(coords) * ATOMIC_MASS[ion]
        f_ion = atomic_form_factor(ion, s) * np.exp(-b_iso * s2)
        phase = np.exp(2j * np.pi * (hkl @ coords.T))
        F += occ * f_ion * np.sum(phase, axis=1)

    F2 = np.real(F * np.conj(F))
    if abs(pref_orient_r - 1.0) >= 1e-5:
        F2 = F2 * _march_dollase_factor(hkl, pref_orient_r, axis=(1.0, 1.0, 1.0))

    unique_q, inv = np.unique(h2k2l2, return_inverse=True)
    sum_F2 = np.bincount(inv, weights=F2)

    keep = sum_F2 > 1e-4
    unique_q, sum_F2 = unique_q[keep], sum_F2[keep]

    d_u = a / np.sqrt(unique_q)
    sin_u = wavelength / (2.0 * d_u)
    theta_u = np.arcsin(sin_u)
    two_theta_u = np.degrees(2.0 * theta_u)

    cos2t = np.cos(2.0 * theta_u)
    lp = (1.0 + cos2t * cos2t) / (sin_u * sin_u * np.cos(theta_u))

    V = a ** 3
    NA = 6.02214076e23
    rho = cell_mass_amu / (NA * V * 1e-24)

    intensity_per_mass = (sum_F2 * lp) / (rho * V * V)
    return two_theta_u, intensity_per_mass, rho


def pseudo_voigt_profile(
    two_theta_grid: np.ndarray,
    peak_positions: np.ndarray,
    peak_intensities: np.ndarray,
    crystallite_nm: float,
    microstrain: float = 0.0005,
    u_inst: float = 0.009,
    v_inst: float = -0.0035,
    w_inst: float = 0.0085,
    eta: float = 0.45,
    wavelength: float = LAMBDA_KA1,
) -> np.ndarray:
    """Area-normalized Pseudo-Voigt profile with benchtop Caglioti + Scherrer + microstrain FWHM."""
    y = np.zeros_like(two_theta_grid)
    if len(peak_positions) == 0:
        return y

    theta_rad = np.radians(peak_positions / 2.0)
    tan_t = np.tan(theta_rad)
    cos_t = np.cos(theta_rad)

    fwhm_inst_sq = np.maximum(u_inst * tan_t ** 2 + v_inst * tan_t + w_inst, 0.002)
    beta_size = np.degrees(0.9 * wavelength / (crystallite_nm * 10.0 * cos_t))
    beta_strain = np.degrees(4.0 * microstrain * tan_t)
    fwhm = np.sqrt(fwhm_inst_sq + beta_size ** 2 + beta_strain ** 2)

    sigma = fwhm / (2.0 * math.sqrt(2.0 * math.log(2.0)))
    gamma = fwhm / 2.0

    dt = two_theta_grid[:, None] - peak_positions[None, :]
    gauss = (1.0 / (sigma[None, :] * math.sqrt(2.0 * math.pi))) * np.exp(-0.5 * (dt / sigma[None, :]) ** 2)
    lorentz = (gamma[None, :] / math.pi) / (dt * dt + gamma[None, :] ** 2)
    pv = (1.0 - eta) * gauss + eta * lorentz
    return pv @ peak_intensities


@dataclass
class SimulationParams:
    rock_salt_wt_pct: float
    ordered_frac: float
    li_ratio: float = 1.00
    ni_ratio: float = 0.50
    mn_ratio: float = 1.50
    a_spinel_dis: float = 8.176
    a_spinel_ord: float = 8.167
    a_rock_salt: float = 4.152
    crystallite_spinel_nm: float = 85.0
    crystallite_rs_nm: float = 42.0
    microstrain: float = 0.0006
    u_o: float = 0.263
    tm_8a_occ: float = 0.005
    pref_orient_r: float = 1.0
    rs_li_frac: float = 0.22
    zero_shift_deg: float = 0.0
    displacement_mm: float = 0.0     # v2: sample-height displacement (mm), R = 150 mm
    bg_over_400: float = 0.0         # v2: flat fluorescence background / spinel (400) net height
    eta: float = 0.45
    scale_counts: float = 6000.0     # v2: 2-min scan strongest peak counts (4000-8000)
    bg_level: float = 280.0
    add_noise: bool = True


class LNMOXRDSimulator:
    """Fast 3-phase Cu Ka1/Ka2 XRD simulator calibrated to benchtop Mn-spinel measurements."""

    def __init__(self, two_theta_min: float = 10.0, two_theta_max: float = 90.0, step: float = 0.02):
        n_pts = int(round((two_theta_max - two_theta_min) / step)) + 1
        self.two_theta = np.linspace(two_theta_min, two_theta_max, n_pts)

    def _simulate_single_phase(
        self,
        sites: List[Tuple[str, float, float, np.ndarray]],
        a: float,
        crystallite_nm: float,
        microstrain: float,
        zero_shift_deg: float,
        displacement_mm: float,
        eta: float,
        pref_orient_r: float = 1.0,
    ) -> np.ndarray:
        pos1, int1, _ = compute_phase_peaks(sites, a, 10.0, 90.0, LAMBDA_KA1, pref_orient_r=pref_orient_r)
        y1 = pseudo_voigt_profile(
            self.two_theta,
            pos1 + zero_shift_deg + disp_deg(pos1, displacement_mm),
            int1,
            crystallite_nm,
            microstrain=microstrain,
            eta=eta,
            wavelength=LAMBDA_KA1,
        )
        pos2, int2, _ = compute_phase_peaks(sites, a, 10.0, 90.0, LAMBDA_KA2, pref_orient_r=pref_orient_r)
        y2 = pseudo_voigt_profile(
            self.two_theta,
            pos2 + zero_shift_deg + disp_deg(pos2, displacement_mm),
            int2 * KA2_RATIO,
            crystallite_nm,
            microstrain=microstrain,
            eta=eta,
            wavelength=LAMBDA_KA2,
        )
        return y1 + y2

    def simulate(self, params: SimulationParams, rng: np.random.Generator | None = None) -> Dict[str, np.ndarray]:
        if rng is None:
            rng = np.random.default_rng(0)

        w_rs = float(np.clip(params.rock_salt_wt_pct / 100.0, 0.0, 0.5))
        w_spinel = 1.0 - w_rs
        f_ord = float(np.clip(params.ordered_frac, 0.0, 1.0))
        w_ord = w_spinel * f_ord
        w_dis = w_spinel * (1.0 - f_ord)

        sites_dis = build_fd3m_sites(
            u_o=params.u_o, tm_8a_occ=params.tm_8a_occ,
            li_ratio=params.li_ratio, ni_ratio=params.ni_ratio, mn_ratio=params.mn_ratio,
        )
        sites_ord = build_p4332_sites(
            u_shift=(params.u_o - 0.263) * 0.5, order_degree=0.85 + 0.15 * f_ord,
            li_ratio=params.li_ratio, ni_ratio=params.ni_ratio, mn_ratio=params.mn_ratio,
        )
        sites_rs = build_rocksalt_sites(li_fraction=params.rs_li_frac)

        y_dis_raw = self._simulate_single_phase(
            sites_dis, params.a_spinel_dis, params.crystallite_spinel_nm, params.microstrain,
            params.zero_shift_deg, params.displacement_mm, params.eta, pref_orient_r=params.pref_orient_r
        )
        y_ord_raw = self._simulate_single_phase(
            sites_ord, params.a_spinel_ord, params.crystallite_spinel_nm * 1.08, params.microstrain * 0.75,
            params.zero_shift_deg, params.displacement_mm, params.eta, pref_orient_r=params.pref_orient_r
        )
        y_rs_raw = self._simulate_single_phase(
            sites_rs, params.a_rock_salt, params.crystallite_rs_nm, params.microstrain * 1.3,
            params.zero_shift_deg, params.displacement_mm, params.eta, pref_orient_r=1.0
        )

        c_dis = w_dis * y_dis_raw
        c_ord = w_ord * y_ord_raw
        c_rs = w_rs * y_rs_raw
        signal_raw = c_dis + c_ord + c_rs

        norm = params.scale_counts / max(float(np.max(signal_raw)), 1e-12)
        c_dis *= norm
        c_ord *= norm
        c_rs *= norm

        # v2 fluorescence background scaled to spinel (400) net height + scaled residual low-angle term
        tt = self.two_theta
        m400 = (tt > 43.8) & (tt < 44.8)
        h400 = float((c_dis + c_ord + c_rs)[m400].max())
        fluo = params.bg_over_400 * h400
        old_bg = (
            params.bg_level
            + 520.0 * np.exp(-(tt - 10.0) / 8.5)
            + 95.0 * np.exp(-0.5 * ((tt - 22.0) / 7.5) ** 2)
            - 0.75 * (tt - 10.0)
        )
        bg = fluo + (params.scale_counts / 52000.0) * np.maximum(old_bg, 40.0)

        expected = bg + c_dis + c_ord + c_rs
        if params.add_noise:
            counts = rng.poisson(np.maximum(expected, 1.0)).astype(np.float64)
        else:
            counts = expected.copy()

        return {
            "two_theta": self.two_theta.copy(),
            "counts": counts,
            "expected": expected,
            "background": bg,
            "disordered": c_dis,
            "ordered": c_ord,
            "rock_salt": c_rs,
        }
