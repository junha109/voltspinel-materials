"""
Baseline Search-Match Evaluator for VoltSpinel LNMO QC XRD v2 dataset (2-min routine scans).
"""

from __future__ import annotations

import argparse
import json
from pathlib import Path

import numpy as np
import pandas as pd
from scipy.signal import find_peaks

DB_NIO_LINES = [37.25, 43.28, 62.87]
DB_P4332_SUPER_LINES = [15.33, 24.36, 26.72]


def evaluate_baseline(data_dir: Path, pattern_subdir: str = "patterns_2min") -> None:
    labels = pd.read_csv(data_dir / "labels.csv")
    patterns_dir = data_dir / pattern_subdir

    peak_cache = {}
    for bid in labels["batch_id"]:
        arr = np.loadtxt(patterns_dir / f"{bid}.xy")
        tt, y = arr[:, 0], arr[:, 1]
        bg_est = np.percentile(y, 15)
        y_net = np.maximum(y - bg_est, 0.0)
        peak_cache[bid] = (tt, y_net, float(np.max(y_net)))

    configs = [(0.1, 2.0), (0.2, 2.0), (0.4, 2.0), (0.2, 0.5), (0.4, 0.5)]
    summary_rows = []
    for win_deg, thresh_pct in configs:
        rs_detected, ord_detected = [], []
        for bid in labels["batch_id"]:
            tt, y_net, y_max = peak_cache[bid]
            p_idx, _ = find_peaks(y_net, prominence=y_max * (thresh_pct / 100.0), distance=8)
            p_pos = tt[p_idx]
            rs_detected.append(any(np.any(np.abs(p_pos - ref) <= win_deg) for ref in DB_NIO_LINES))
            ord_detected.append(any(np.any(np.abs(p_pos - ref) <= win_deg) for ref in DB_P4332_SUPER_LINES))

        rs_det = np.array(rs_detected, dtype=bool)
        ord_det = np.array(ord_detected, dtype=bool)

        is_fail = ~labels["rs_pass"].values
        is_bound = labels["is_boundary_fail"].values
        is_pass = labels["rs_pass"].values
        is_ord = (labels["product_type"].values == "VS-5O")

        summary_rows.append({
            "window_deg": win_deg,
            "threshold_pct": thresh_pct,
            "fail_miss_pct": round(100.0 * float(np.mean(~rs_det[is_fail])), 1),
            "boundary_miss_pct": round(100.0 * float(np.mean(~rs_det[is_bound])), 1),
            "pass_fp_pct": round(100.0 * float(np.mean(rs_det[is_pass])), 1),
            "ordered_recall_pct": round(100.0 * float(np.mean(ord_det[is_ord])), 1),
            "ordered_fp_pct": round(100.0 * float(np.mean(ord_det[~is_ord])), 1),
        })

    res_df = pd.DataFrame(summary_rows)
    res_df.to_csv(data_dir / "baseline_results.csv", index=False)
    with open(data_dir / "baseline_summary.json", "w", encoding="utf-8") as f:
        json.dump(summary_rows, f, indent=2, ensure_ascii=False)
    print(res_df.to_string(index=False))


if __name__ == "__main__":
    parser = argparse.ArgumentParser()
    parser.add_argument("--data", type=str, default="../data")
    parser.add_argument("--subdir", type=str, default="patterns_2min")
    args = parser.parse_args()
    evaluate_baseline(Path(args.data), args.subdir)
